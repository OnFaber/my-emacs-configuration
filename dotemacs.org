Configurazione Emacs literate. Per generare =dotemacs.el=:
=M-x org-babel-tangle= (=C-c C-v t=).

Funziona su Emacs 29.3+ (Ubuntu) ed Emacs 30 (Arch). Dove le due
versioni si comportano diversamente c'è un controllo su
=emacs-major-version=. I percorsi partono da =user-emacs-directory=,
quindi la stessa config va bene in =~/.config/emacs= e in =~/.emacs.d=.

* Early Init

Il file =early-init.el= viene caricato prima di =init.el=.
Disabilita GUI elements e =package.el= al boot.

#+begin_src emacs-lisp :tangle early-init.el
;;; early-init.el --- -*- no-byte-compile: t; lexical-binding: t; -*-
(setq package-enable-at-startup nil)
(customize-set-variable 'menu-bar-mode nil)
(customize-set-variable 'scroll-bar-mode nil)
(customize-set-variable 'tool-bar-mode nil)
(customize-set-variable 'ring-bell-function 'ignore)
;; Gli avvisi della compilazione nativa riguardano il codice dei pacchetti,
;; non la config: si scrivono in *Warnings* senza aprire la finestra
(setq native-comp-async-report-warnings-errors 'silent)
(provide 'early-init-file)
;;; early-init.el ends here
#+end_src

* Init

Il file =init.el= carica i pacchetti e poi questo file.

#+begin_src emacs-lisp :tangle init.el
;;; init.el --- -*- no-byte-compile: t; lexical-binding: t; -*-

(require 'package)
(setq package-archives
      '(("gnu"    . "https://elpa.gnu.org/packages/")
        ("nongnu" . "https://elpa.nongnu.org/nongnu/")
        ("melpa"  . "https://melpa.org/packages/")))
(setq package-archive-priorities
      '(("melpa" . 4) ("nongnu" . 2) ("gnu" . 1)))
;; Permette di aggiornare i pacchetti inclusi in Emacs (eglot, transient):
;; su Emacs 29 quelli inclusi sono troppo vecchi.
;; Una volta: M-x package-upgrade RET eglot, poi transient.
(setq package-install-upgrade-built-in t)
(package-initialize)

(unless (package-installed-p 'use-package)
  (package-refresh-contents)
  (package-install 'use-package))
(require 'use-package)
(setq use-package-always-ensure t)

;; Paths
(defconst my/cache-directory
  (expand-file-name ".cache/" user-emacs-directory))
(defconst my/custom-directory
  (expand-file-name "core/custom/" user-emacs-directory))
(defconst my/libs-directory
  (expand-file-name "core/libs/" user-emacs-directory))

(make-directory my/cache-directory 'parents)
(add-to-list 'load-path my/custom-directory)

(dolist (subdir '("vue-ts-mode/" "peek/"))
  (let ((path (expand-file-name subdir my/libs-directory)))
    (when (file-exists-p path)
      (add-to-list 'load-path path))))

;; Transparency
;; (set-frame-parameter nil 'alpha-background 91)
;; (add-to-list 'default-frame-alist '(alpha-background . 91))

;; Le impostazioni salvate da Customize (e da package.el) vanno qui,
;; non in init.el, che viene rigenerato dal tangle
(setq custom-file (expand-file-name "custom.el" user-emacs-directory))
(load custom-file 'noerror)

;; Load main config
(load (expand-file-name "dotemacs.el" user-emacs-directory))

(provide 'init)
;;; init.el ends here
#+end_src

(defun my/posframe-refposhandler-exwm (&optional _frame)
  "Origine (x . y) dello schermo del workspace EXWM attivo.
Restituisce nil fuori da EXWM: posframe usa allora un frame figlio."
  (when-let* (((bound-and-true-p exwm--connection))
              ((frame-live-p exwm-workspace--current))
              (geom (frame-parameter exwm-workspace--current 'exwm-geometry)))
    (cons (slot-value geom 'x) (slot-value geom 'y))))

  "Inserisce l'import PKG in cima al buffer se non esiste."

#+begin_src emacs-lisp

;;; --- Vai alla definizione anche nelle librerie .NET ------------------------
;; Con `--features metadata-uris', csharp-ls risponde alle definizioni delle
;; librerie compilate con indirizzi "csharp:/..." invece che con file.
;; Eglot passa ogni indirizzo da `eglot-uri-to-path': qui intercettiamo
;; quelli "csharp:/", chiediamo il sorgente decompilato al server
;; (richiesta `csharp/metadata'), lo salviamo in cache e restituiamo il
;; percorso di quel file. Da lì in poi xref lo apre come un file normale.

(defconst my/csharp-metadata-dir
  (expand-file-name "csharp-metadata/" my/cache-directory)
  "Dove salvare i sorgenti decompilati da csharp-ls.")

(defun my/csharp-metadata--path (uri)
  "Percorso in cache per l'indirizzo csharp:/ URI."
  (let ((rel (string-remove-prefix "csharp:/" (url-unhex-string uri))))
    ;; niente caratteri strani e niente ".." nel nome
    (setq rel (replace-regexp-in-string "[^[:alnum:]._/-]" "_" rel))
    (setq rel (replace-regexp-in-string "\\.\\.+" "_" rel))
    (expand-file-name rel my/csharp-metadata-dir)))

(defun my/csharp-metadata--fetch (uri)
  "Restituisce il file con il sorgente decompilato di URI, scaricandolo se serve."
  (let ((path (my/csharp-metadata--path uri)))
    (if (file-exists-p path)
        path
      (when-let* ((server (eglot-current-server))
                  (res (jsonrpc-request server :csharp/metadata
                                        `(:textDocument (:uri ,uri))))
                  (source (plist-get res :source)))
        (make-directory (file-name-directory path) t)
        (with-temp-file path (insert source))
        path))))

(defun my/csharp-metadata--uri-to-path (orig uri)
  "Avvolge `eglot-uri-to-path' (ORIG) per gestire gli indirizzi csharp:/."
  (let ((s (if (keywordp uri) (substring (symbol-name uri) 1) uri)))
    (or (and (stringp s)
             (string-prefix-p "csharp:/" s)
             (condition-case err
                 (my/csharp-metadata--fetch s)
               (error (message "csharp-ls: sorgente non disponibile (%s)"
                               (error-message-string err))
                      nil)))
        (funcall orig uri))))

(with-eval-after-load 'eglot
  (advice-add 'eglot-uri-to-path :around #'my/csharp-metadata--uri-to-path))

(defun my/csharp-metadata--read-only ()
  "Apre in sola lettura i sorgenti decompilati: modificarli non avrebbe senso."
  (when (and buffer-file-name
             (string-prefix-p my/csharp-metadata-dir buffer-file-name))
    (read-only-mode 1)))

(add-hook 'find-file-hook #'my/csharp-metadata--read-only)

;; Tutte ereditano dalle facce del tema: cambiano colore con lui.

(defface my/csharp-doc-face

  "Testo della documentazione (summary, param, returns...).")

(defface my/csharp-doc-remarks-face
  '((t :inherit (shadow my/csharp-doc-face) :slant italic))
  "Testo delle sezioni <remarks>: più discreto del summary.")

(defface my/csharp-doc-bar-face
  '((t :inherit font-lock-comment-delimiter-face))
  "Barra verticale mostrata al posto di ///.")

(defface my/csharp-doc-label-face
  '((t :inherit font-lock-keyword-face :weight bold :slant normal))
  "Etichette: Restituisce, Valore, Eccezione, punti elenco.")

(defface my/csharp-doc-param-face
  '((t :inherit font-lock-variable-name-face :weight bold :slant normal))
  "Nomi dei parametri (<param>, <paramref>).")

(defface my/csharp-doc-ref-face
  '((t :inherit font-lock-type-face :slant normal))
  "Riferimenti a tipi e membri (<see cref=\"...\"/>).")

(defface my/csharp-doc-keyword-face
  '((t :inherit font-lock-keyword-face :slant normal))
  "Parole chiave (<see langword=\"null\"/>).")

(defface my/csharp-doc-code-face
  '((t :inherit font-lock-constant-face :slant normal))
  "Codice in linea (<c>...</c>) e blocchi <code>.")


;;; --- Analisi dei tag --------------------------------------------------------

(defconst my/csharp-doc--tag-re
  "<\\(/?\\)\\([A-Za-z]+\\)\\([^>]*?\\)\\(/?\\)>"
  "Tag XML: 1 = \"/\" se di chiusura, 2 = nome, 3 = attributi, 4 = \"/\" se autochiuso.")

(defconst my/csharp-doc--sections
  '("summary" "remarks" "returns" "value" "param" "typeparam" "exception" "example")
  "Tag che aprono una sezione della documentazione.")

(defconst my/csharp-doc--block-line-re
  "\\`[ \t]*<\\(/?\\)\\(summary\\|remarks\\|code\\|list\\|para\\)\\b[^>]*>[ \t]*\\'"
  "Riga che contiene solo un tag di blocco: viene nascosta del tutto.")

(defun my/csharp-doc--attr (attrs)
  "Coppia (TIPO . VALORE) di cref/name/langword in ATTRS, senza prefisso \"T:\"."
  (when (string-match
         "\\(cref\\|name\\|langword\\)=\"\\(?:[A-Z]:\\)?\\([^\"]+\\)\"" attrs)
    (cons (match-string 1 attrs) (match-string 2 attrs))))

(defun my/csharp-doc--tag-display (tag closing attrs)
  "Testo da mostrare al posto del tag TAG; nil per nasconderlo."
  (pcase-let ((`(,tipo . ,nome) (my/csharp-doc--attr attrs)))
    (cond
     (closing nil)
     ((member tag '("see" "seealso"))
      (and nome (propertize nome 'face (if (equal tipo "langword")
                                           'my/csharp-doc-keyword-face
                                         'my/csharp-doc-ref-face))))
     ((member tag '("paramref" "typeparamref"))
      (and nome (propertize nome 'face 'my/csharp-doc-param-face)))
     ((member tag '("param" "typeparam"))
      (and nome (concat (propertize nome 'face 'my/csharp-doc-param-face)
                        (propertize ": " 'face 'my/csharp-doc-face))))
     ((string= tag "returns")
      (propertize "Restituisce: " 'face 'my/csharp-doc-label-face))
     ((string= tag "value")
      (propertize "Valore: " 'face 'my/csharp-doc-label-face))
     ((string= tag "exception")
      (concat (propertize "Eccezione " 'face 'my/csharp-doc-label-face)
              (propertize (or nome "") 'face 'my/csharp-doc-ref-face)
              (propertize ": " 'face 'my/csharp-doc-label-face)))
     ((string= tag "item")
      (propertize "• " 'face 'my/csharp-doc-label-face))
     (t nil))))                          ; <c>, </c>, <para>, <list>... → nascosti


;;; --- Overlay ----------------------------------------------------------------

(defun my/csharp-doc--ov (beg end &rest props)
  "Crea un overlay di `my/csharp-doc-mode' tra BEG ed END con PROPS."
  (let ((ov (make-overlay beg end)))
    (overlay-put ov 'my-csharp-doc t)
    (overlay-put ov 'evaporate t)
    (while props
      (overlay-put ov (pop props) (pop props)))
    ov))

(defun my/csharp-doc--segment (beg end section in-code)
  "Colora il testo tra BEG ed END secondo SECTION e IN-CODE."
  (when (< beg end)
    (my/csharp-doc--ov beg end 'face
                       (cond (in-code 'my/csharp-doc-code-face)
                             ((equal section "remarks") 'my/csharp-doc-remarks-face)
                             (t 'my/csharp-doc-face)))))

(defun my/csharp-doc--block-bounds (beg end)
  "Estende BEG ed END fino ai bordi dei blocchi /// che li contengono.
Serve perché lo stile di una riga dipende dai tag aperti nelle righe sopra."
  (save-excursion
    (goto-char beg)
    (beginning-of-line)
    (while (and (not (bobp))
                (save-excursion (forward-line -1) (looking-at-p "[ \t]*///")))
      (forward-line -1))
    (setq beg (point))
    (goto-char end)
    (end-of-line)
    (while (and (not (eobp))
                (save-excursion (forward-line 1) (looking-at-p "[ \t]*///")))
      (forward-line 1)
      (end-of-line))
    (cons beg (point))))


;;; --- Resa -------------------------------------------------------------------

(defun my/csharp-doc--render (beg end)
  "Rende i commenti /// tra BEG ed END. Chiamata da jit-lock."
  (save-excursion
    (save-match-data
      (pcase-let ((`(,beg . ,end) (my/csharp-doc--block-bounds beg end)))
        (remove-overlays beg (min (point-max) (1+ end)) 'my-csharp-doc t)
        (goto-char beg)
        (let ((section nil)   ; sezione aperta: "summary", "remarks", "param"...
              (in-code nil))  ; dentro <c> o <code>?
          (while (re-search-forward "^[ \t]*\\(///[ \t]?\\)\\(.*\\)$" end t)
            (let ((pref-beg (match-beginning 1)) (pref-end (match-end 1))
                  (txt-beg  (match-beginning 2)) (txt-end  (match-end 2))
                  (testo    (match-string 2))
                  (riga-beg (line-beginning-position))
                  (riga-end (min (point-max) (1+ (line-end-position)))))
              ;; nuovo blocco di commenti: si riparte da zero
              (unless (save-excursion
                        (forward-line -1)
                        (and (< (point) riga-beg) (looking-at-p "[ \t]*///")))
                (setq section nil in-code nil))
              (if (string-match my/csharp-doc--block-line-re testo)
                  ;; riga con solo <summary>, </remarks>, <code>...: sparisce
                  (let ((closing (string= (match-string 1 testo) "/"))
                        (tag (match-string 2 testo)))
                    (my/csharp-doc--ov riga-beg riga-end 'invisible t)
                    (pcase tag
                      ("code" (setq in-code (not closing)))
                      ((or "summary" "remarks") (setq section (unless closing tag)))))
                ;; "///" diventa una barra verticale
                (my/csharp-doc--ov pref-beg pref-end
                                   'display (propertize "│ " 'face 'my/csharp-doc-bar-face))
                ;; testo e tag in linea, un pezzo alla volta
                (let ((pos txt-beg))
                  (save-excursion
                    (goto-char txt-beg)
                    (while (re-search-forward my/csharp-doc--tag-re txt-end t)
                      (let* ((tb (match-beginning 0))
                             (te (match-end 0))
                             (closing (string= (match-string 1) "/"))
                             (tag (match-string 2))
                             (attrs (match-string 3))
                             (self-closing (string= (match-string 4) "/"))
                             (disp (my/csharp-doc--tag-display tag closing attrs)))
                        (my/csharp-doc--segment pos tb section in-code)
                        (unless self-closing
                          (cond ((member tag '("c" "code"))
                                 (setq in-code (not closing)))
                                ((member tag my/csharp-doc--sections)
                                 (setq section (unless closing tag)))))
                        (if disp
                            (my/csharp-doc--ov tb te 'display disp)
                          (my/csharp-doc--ov tb te 'invisible t))
                        (setq pos te))))
                  (my/csharp-doc--segment pos txt-end section in-code))))))))))

        (jit-lock-refontify))
    (jit-lock-unregister #'my/csharp-doc--render)))

;;; --- HTML e Twig colorati con font-lock -------------------------------------
;; php-mode colora solo il PHP: queste regole aggiungono tag HTML, attributi
;; e sintassi Twig. I colori vengono dal tema.
;; Servono con php-mode (Emacs 29) e per i file .twig; php-ts-mode
;; (Emacs 30) colora già l'HTML da solo e non usa queste regole.

(defface my/html-tag-face     '((t :inherit font-lock-function-name-face)) "Nome del tag.")
(defface my/html-attr-face    '((t :inherit font-lock-variable-name-face)) "Attributo.")
(defface my/html-bracket-face '((t :inherit shadow)) "< </ > />")
(defface my/twig-delim-face   '((t :inherit font-lock-preprocessor-face :weight bold)) "{{ }} {% %}")
(defface my/twig-keyword-face '((t :inherit font-lock-keyword-face :weight bold)) "if, for, block...")
(defface my/twig-name-face    '((t :inherit font-lock-function-name-face)) "Nome di block/macro.")
(defface my/twig-filter-face  '((t :inherit font-lock-builtin-face)) "|upper, |date...")

(defconst my/html-tags
  '("a" "abbr" "article" "aside" "audio" "b" "body" "br" "button" "canvas" "code"
    "dd" "div" "dl" "dt" "em" "fieldset" "footer" "form" "h1" "h2" "h3" "h4" "h5" "h6"
    "head" "header" "hr" "html" "i" "iframe" "img" "input" "label" "legend" "li" "link"
    "main" "meta" "nav" "ol" "option" "p" "pre" "script" "section" "select" "small"
    "span" "strong" "style" "svg" "table" "tbody" "td" "template" "textarea" "tfoot"
    "th" "thead" "title" "tr" "ul" "video")
  "Tag riconosciuti: solo nomi veri, così `$a<$b' non viene colorato.")

(defconst my/twig-operators
  '("and" "or" "not" "in" "is" "with" "only" "ignore" "missing" "as"
    "matches" "starts" "ends" "b-and" "b-or" "b-xor" "if" "else")
  "Operatori e parole riservate dentro {{ }} e {% %}.")

(defconst my/twig-constants
  '("true" "false" "null" "none" "empty" "defined" "even" "odd"
    "iterable" "same" "divisible" "by" "loop" "_self")
  "Costanti, test (is empty, is defined...) e variabili speciali.")

(defun my/twig--region-end ()
  "Dalla fine di {{ o {%: posizione della chiusura }} o %} sulla stessa riga.
Serve da limite alle regole che valgono solo dentro i tag Twig."
  (goto-char (match-end 0))
  (save-excursion
    (save-match-data              ; le regole successive ripartono dallo stesso {{
      (if (re-search-forward "-?[}%]}" (line-end-position) t)
          (match-beginning 0)
        (line-end-position)))))

(defun my/twig--inside (regexp face &optional group)
  "Regola: colora con FACE il GROUP (default 0) di REGEXP, solo dentro {{ }} e {% %}."
  `(,regexp (my/twig--region-end) nil (,(or group 0) ',face t)))

(defconst my/html-twig-keywords
  `(;; ── HTML ──────────────────────────────────────────────
    ;; <div   </div
    (,(concat "\\(</?\\)" (regexp-opt my/html-tags 'words))
     (1 'my/html-bracket-face t)
     (2 'my/html-tag-face t))
    ;; >  />  (ma non -> => ?> %} }})
    ("[^-=?%}]\\(/?>\\)" 1 'my/html-bracket-face t)
    ;; attributi: class="..."  data-id='...'  @click="..."
    ("[ \t\n]\\([[:alpha:]@:][[:alnum:]_:.@-]*\\)=[\"']" 1 'my/html-attr-face t)

    ;; ── Twig: dentro {{ }} e {% %} ──────────────────────
    ;; Le regole sono in ordine: quelle dopo vincono su quelle prima.
    ("{[{%]-?"
     ;; variabili
     ,(my/twig--inside "[[:alpha:]_][[:alnum:]_]*" 'font-lock-variable-use-face)
     ;; .attributo
     ,(my/twig--inside "\\.\\([[:alpha:]_][[:alnum:]_]*\\)" 'font-lock-property-use-face 1)
     ;; chiave: valore  (negli hash { chiave: valore })
     ,(my/twig--inside "\\([[:alpha:]_][[:alnum:]_]*\\)[ \t]*:" 'font-lock-property-name-face 1)
     ;; funzione(
     ,(my/twig--inside "\\([[:alpha:]_][[:alnum:]_]*\\)[ \t]*(" 'font-lock-function-call-face 1)
     ;; |filtro
     ,(my/twig--inside "|[ \t]*\\([[:alpha:]_][[:alnum:]_]*\\)" 'my/twig-filter-face 1)
     ;; operatori a parole: and, or, not, is, in, with...
     ,(my/twig--inside (regexp-opt my/twig-operators 'words) 'font-lock-keyword-face)
     ;; true, false, null, empty, defined...
     ,(my/twig--inside (regexp-opt my/twig-constants 'words) 'font-lock-constant-face)
     ;; numeri
     ,(my/twig--inside "\\<[0-9]+\\(?:\\.[0-9]+\\)?\\>" 'font-lock-number-face)
     ;; operatori a simboli: ~ == != <= >= .. ?: ?? + - * /
     ,(my/twig--inside "\\(?:[=!<>]=\\|\\.\\.\\|\\?[?:]?\\|[~+*/<>]\\)" 'font-lock-operator-face)
     ;; 'stringhe' e "stringhe": per ultime, così vincono su tutto
     ,(my/twig--inside "'[^']*'\\|\"[^\"]*\"" 'font-lock-string-face))

    ;; parola chiave del tag: {% if   {% block   {% endfor
    ("{%-?[ \t]*\\([[:alpha:]_]+\\)" 1 'my/twig-keyword-face t)
    ;; nome dopo block/endblock/macro: {% block body %}
    ("{%-?[ \t]*\\(?:end\\)?\\(?:block\\|macro\\)[ \t]+\\([[:alnum:]_]+\\)"
     1 'my/twig-name-face t)
    ;; delimitatori
    ("{{-?\\|-?}}\\|{%-?\\|-?%}" 0 'my/twig-delim-face t)
    ;; {# commenti #}: per ultimi
    ("{#.*?#}" 0 'font-lock-comment-face t)))

  :lighter nil

      (font-lock-add-keywords nil my/html-twig-keywords 'append)
    (font-lock-remove-keywords nil my/html-twig-keywords))
  (font-lock-flush))
